# SMIB desktop simulator: the firmware's app, CMS, SAS, mux and UI code with
# LVGL, built for macOS/Linux. SDL stands in for the display + touch, the
# keyboard for the RC522, a pseudo-terminal for UART0 and the host's audio
# output for the speaker. See "Testing without hardware" in ../README.md.
#
#   cmake -S sim -B sim/build && cmake --build sim/build -j
cmake_minimum_required(VERSION 3.20)
# The SDK that matches the selected developer tools (`xcrun --sdk macosx`).
# CMake's default lookup can pick a newer Command Line Tools SDK that Xcode's
# linker can't read ("tapi error: malformed file").
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_OSX_SYSROOT AND NOT DEFINED ENV{SDKROOT})
  set(CMAKE_OSX_SYSROOT macosx)
endif()
project(smib_sim C)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Debug)
endif()
set(SMIB_SIM_SANITIZE "address" CACHE STRING
    "Sanitizer for the firmware + sim code: address (ASan+UBSan), thread, or none")

get_filename_component(FW_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(LVGL_DIR "${FW_DIR}/managed_components/lvgl__lvgl")
set(SDKCONFIG "${FW_DIR}/sdkconfig")

if(NOT EXISTS "${LVGL_DIR}/lvgl.h" OR NOT EXISTS "${SDKCONFIG}")
  message(FATAL_ERROR
    "The sim builds against the firmware's LVGL (managed_components/) and "
    "sdkconfig, which ESP-IDF generates. Run `idf.py reconfigure` (or a normal "
    "build) once in ${FW_DIR}.")
endif()

# cJSON: the copy ESP-IDF ships, same as the firmware links.
set(IDF_PATH "$ENV{IDF_PATH}" CACHE PATH "ESP-IDF checkout (for cJSON)")
if(NOT IDF_PATH)
  file(GLOB _idf_dirs "$ENV{HOME}/.espressif/v*/esp-idf")
  if(_idf_dirs)
    list(SORT _idf_dirs COMPARE NATURAL ORDER DESCENDING)
    list(GET _idf_dirs 0 _idf)
    set(IDF_PATH "${_idf}" CACHE PATH "ESP-IDF checkout (for cJSON)" FORCE)
  endif()
endif()
set(CJSON_DIR "${IDF_PATH}/components/json/cJSON")
if(NOT EXISTS "${CJSON_DIR}/cJSON.c")
  message(FATAL_ERROR "cJSON not found under IDF_PATH='${IDF_PATH}'. Pass -DIDF_PATH=/path/to/esp-idf")
endif()

# sdkconfig.h from the firmware's sdkconfig: LVGL, log and tick settings only.
# Rewritten only when it changes; editing sdkconfig re-runs this step.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SDKCONFIG}")
file(STRINGS "${SDKCONFIG}" _cfg REGEX "^CONFIG_(LV_|LVGL_|LOG_|FREERTOS_HZ=)")
set(_hdr "// Generated by sim/CMakeLists.txt from ../sdkconfig. Do not edit.\n#pragma once\n")
foreach(_line IN LISTS _cfg)
  if(_line MATCHES "^(CONFIG_[A-Za-z0-9_]+)=(.*)$")
    set(_val "${CMAKE_MATCH_2}")
    if(_val STREQUAL "y")
      set(_val 1)
    endif()
    string(APPEND _hdr "#define ${CMAKE_MATCH_1} ${_val}\n")
  endif()
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/gen/sdkconfig.h" CONTENT "${_hdr}" @ONLY)

find_package(SDL2 REQUIRED)
find_package(Threads REQUIRED)

# LVGL, configured through the firmware's Kconfig values (lvgl/lv_kconfig_sim.h).
file(GLOB_RECURSE LVGL_SRCS CONFIGURE_DEPENDS "${LVGL_DIR}/src/*.c")
add_library(lvgl STATIC ${LVGL_SRCS})
target_include_directories(lvgl PUBLIC
  "${LVGL_DIR}" "${LVGL_DIR}/src" "${CMAKE_BINARY_DIR}/gen" "${CMAKE_CURRENT_SOURCE_DIR}/lvgl")
target_compile_definitions(lvgl PUBLIC
  "LV_CONF_KCONFIG_EXTERNAL_INCLUDE=\"lv_kconfig_sim.h\"" LV_LVGL_H_INCLUDE_SIMPLE)
target_compile_options(lvgl PRIVATE -O2 -w)

add_library(cjson STATIC "${CJSON_DIR}/cJSON.c")
target_include_directories(cjson PUBLIC "${CJSON_DIR}")
target_compile_options(cjson PRIVATE -O2 -w)

# Everything in main/ except the hardware drivers, which src/ replaces.
file(GLOB_RECURSE FW_SRCS CONFIGURE_DEPENDS "${FW_DIR}/main/*.c")
list(FILTER FW_SRCS EXCLUDE REGEX "/main/drivers/")

add_executable(smib_sim
  ${FW_SRCS}
  port/esp.c
  port/freertos.c
  port/uart_pty.c
  src/sim_audio.c
  src/sim_ctl.c
  src/sim_display.c
  src/sim_door.c
  src/sim_main.c
  src/sim_net.c
  src/sim_nfc.c
  src/sim_settings.c)
target_include_directories(smib_sim PRIVATE "${FW_DIR}/main" port/include src)

# Admin passcode and the factory Wi-Fi/CMS defaults, from the same ../.env the
# firmware reads, so the sim's admin panel takes the same code as the board.
set(SMIB_ENV_DIR "${FW_DIR}")
include("${FW_DIR}/cmake/smib_env.cmake")
smib_env_read(SMIB_ENV_DEFS)
target_compile_definitions(smib_sim PRIVATE ${SMIB_ENV_DEFS})
target_link_libraries(smib_sim PRIVATE lvgl cjson SDL2::SDL2 Threads::Threads m)
if(NOT APPLE)
  target_link_libraries(smib_sim PRIVATE util) # openpty
endif()
set_target_properties(smib_sim PROPERTIES C_STANDARD 17 C_EXTENSIONS ON)
target_compile_options(smib_sim PRIVATE -Wall -Wextra -Wno-unused-parameter)

if(SMIB_SIM_SANITIZE STREQUAL "address")
  set(_san -fsanitize=address,undefined -fno-omit-frame-pointer)
elseif(SMIB_SIM_SANITIZE STREQUAL "thread")
  set(_san -fsanitize=thread)
endif()
if(_san)
  target_compile_options(smib_sim PRIVATE ${_san})
  target_link_options(smib_sim PRIVATE ${_san})
endif()
